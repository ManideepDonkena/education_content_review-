# ── helpers ──────────────────────────────────────────────────────────────────

def right_triangle_points(a, b, origin=np.array([0.0, 0.0, 0.0])):
    """Return (A, B, C) where right angle is at B.
    A = origin, B = origin+(a,0,0), C = origin+(a,b,0)"""
    A = np.array([origin[0],        origin[1],        0.0])
    B = np.array([origin[0] + a,    origin[1],        0.0])
    C = np.array([origin[0] + a,    origin[1] + b,    0.0])
    return A, B, C

def make_right_angle_mark(corner, v1, v2, size=0.18, color=INK):
    """Small square marker at corner, legs toward v1 and v2."""
    d1 = (v1 - corner) / np.linalg.norm(v1 - corner) * size
    d2 = (v2 - corner) / np.linalg.norm(v2 - corner) * size
    p1 = corner + d1
    p2 = corner + d1 + d2
    p3 = corner + d2
    return Polygon(corner, p1, p2, p3,
                   stroke_color=color, stroke_width=1.5, fill_opacity=0)

def make_grid_square(origin, side_vec, perp_vec, n, color, fill_color, fill_opacity=0.35):
    """Build an n×n grid square. origin is one corner, side_vec and perp_vec
    are the two edge vectors (length = side length)."""
    lines = VGroup()
    for i in range(n + 1):
        t = i / n
        p0 = origin + t * side_vec
        p1 = p0 + perp_vec
        lines.add(Line(p0, p1, stroke_color=color, stroke_width=0.8))
        p0b = origin + t * perp_vec
        p1b = p0b + side_vec
        lines.add(Line(p0b, p1b, stroke_color=color, stroke_width=0.8))
    corners = [origin, origin + side_vec,
               origin + side_vec + perp_vec, origin + perp_vec]
    sq = Polygon(*corners, stroke_color=color, stroke_width=2,
                 fill_color=fill_color, fill_opacity=fill_opacity)
    return VGroup(sq, lines)

# C02 – LENGTH BECOMES AREA
# ═══════════════════════════════════════════════════════════════════════════
class C02(LessonScene):
    def construct(self):
        self.chapter(2)

        # Triangle geometry (same scale as C01)
        SCALE = 0.55
        OX, OY = -4.8, -1.6
        A = np.array([OX,            OY,           0.0])
        B = np.array([OX + 4*SCALE,  OY,           0.0])
        C = np.array([OX + 4*SCALE,  OY + 3*SCALE, 0.0])

        def make_triangle():
            tri = Polygon(A, B, C,
                          stroke_color=INK, stroke_width=2,
                          fill_color=BG, fill_opacity=0)
            tb = Line(A, B, color=TEAL, stroke_width=3)
            tr = Line(B, C, color=GOLD, stroke_width=3)
            td = Line(A, C, color=PINK, stroke_width=3)
            ra = make_right_angle_mark(B, A, C, size=0.16, color=INK)
            return VGroup(tri, tb, tr, td, ra)

        tri_grp = make_triangle()

        # ── BEAT 05 ──────────────────────────────────────────────────────
        self.begin("05")

        self.play(FadeIn(tri_grp), run_time=0.6)

        # Teal square on bottom leg (outward = downward from A-B)
        # A=(OX,OY), B=(OX+4s,OY); outward = DOWN
        s4 = 4 * SCALE
        s3 = 3 * SCALE
        sq_a_origin = A + np.array([0, -s4, 0])
        sq_a_side   = np.array([s4, 0, 0])
        sq_a_perp   = np.array([0,  s4, 0])
        sq_a = make_grid_square(sq_a_origin, sq_a_side, sq_a_perp,
                                4, TEAL, TEAL, 0.30)

        self.cue("Let us build a square on each side of our triangle",
                 FadeIn(tri_grp), run_time=0.5)

        self.cue("On the four-metre teal leg, we grow a square outward",
                 Create(sq_a), run_time=1.2)

        lbl_16 = txt("16 sq m", 22, TEAL)
        lbl_16.move_to(sq_a_origin + np.array([s4/2, s4/2, 0]))

        self.cue("Its area is four times four, which is sixteen square metres",
                 FadeIn(lbl_16), run_time=0.8)

        # Gold square on right leg (outward = RIGHT from B-C)
        # B=(OX+4s,OY), C=(OX+4s,OY+3s); outward = RIGHT
        sq_b_origin = B
        sq_b_side   = np.array([s3, 0, 0])
        sq_b_perp   = np.array([0,  s3, 0])
        sq_b = make_grid_square(sq_b_origin, sq_b_side, sq_b_perp,
                                3, GOLD, GOLD, 0.30)

        self.cue("On the three-metre gold leg, we grow another square outward",
                 Create(sq_b), run_time=1.0)

        lbl_9 = txt("9 sq m", 22, GOLD)
        lbl_9.move_to(sq_b_origin + np.array([s3/2, s3/2, 0]))

        self.cue("Its area is three times three, which is nine square metres",
                 FadeIn(lbl_9), run_time=0.7)

        p1 = eq(r"a^2 = 4^2 = 16", 34, TEAL)
        p2 = eq(r"b^2 = 3^2 = 9",  34, GOLD)
        self.cue("These are real geometric squares with real area",
                 self.panel_to(p1, p2), run_time=0.8)

        self.finish()

        # ── BEAT 06 ──────────────────────────────────────────────────────
        self.begin("06")

        # Show 16+9=25
        sum_eq = eq(r"16 + 9 = 25", 34, INK).move_to([3.5, 1.5, 0])
        self.cue("Now add those two areas together",
                 FadeIn(sum_eq), run_time=0.7)

        # Pink square on diagonal (outward = upper-left from A-C)
        # A=(OX,OY), C=(OX+4s,OY+3s)
        # diagonal vector AC, perp outward (rotate 90° CCW)
        AC = C - A
        AC_len = np.linalg.norm(AC)
        AC_unit = AC / AC_len
        perp_unit = np.array([-AC_unit[1], AC_unit[0], 0.0])
        s5 = AC_len  # should be 5*SCALE
        sq_c_origin = A
        sq_c_side   = AC
        sq_c_perp   = perp_unit * s5

        sq_c = make_grid_square(sq_c_origin, sq_c_side, sq_c_perp,
                                5, PINK, PINK, 0.28)

        self.cue("The claim of the Pythagorean theorem is that the square built on the diagonal",
                 Create(sq_c), run_time=1.2)

        lbl_25 = txt("25 sq m", 22, PINK)
        lbl_25.move_to(sq_c_origin + (sq_c_side + sq_c_perp) / 2)

        self.cue("also has an area of exactly twenty-five square metres",
                 FadeIn(lbl_25), run_time=0.7)

        # sqrt arrow
        sqrt_eq = eq(r"\sqrt{25} = 5\text{ m}", 34, PINK).move_to([3.5, 0.5, 0])
        arr_to_sqrt = Arrow([3.5, 1.1, 0], [3.5, 0.9, 0],
                            color=PINK, buff=0.05, stroke_width=2)

        self.cue("the diagonal's length must be the number whose square is twenty-five",
                 AnimationGroup(FadeIn(arr_to_sqrt), FadeIn(sqrt_eq),
                                lag_ratio=0.3),
                 run_time=0.9)

        p3 = eq(r"16+9=25,\; c=5\text{ m}", 30, INK)
        p4 = eq(r"c^2=a^2+b^2", 34, INK)
        self.cue("the area is twenty-five square metres, but the length is five metres",
                 self.panel_to(p3, p4), run_time=0.8)

        self.finish()

        # ── BEAT 07 ──────────────────────────────────────────────────────
        self.begin("07")

        # Replace numeric labels with variable labels on triangle sides
        lbl_a = txt("a", 24, TEAL)
        lbl_b = txt("b", 24, GOLD)
        lbl_c = txt("c", 24, PINK)
        lbl_a.move_to((A + B) / 2 + np.array([0, -0.22, 0]))
        lbl_b.move_to((B + C) / 2 + np.array([0.22, 0, 0]))
        lbl_c.move_to((A + C) / 2 + np.array([-0.22, 0.1, 0]))

        self.cue("Five lies between four and seven, so the answer passes our estimate",
                 AnimationGroup(FadeIn(lbl_a), FadeIn(lbl_b), FadeIn(lbl_c),
                                lag_ratio=0.3),
                 run_time=0.8)

        # Main formula large
        main_formula = eq(r"c^2 = a^2 + b^2", 44, INK).move_to([3.5, 1.3, 0])
        self.cue("Now call the teal leg a, the gold leg b, and the pink diagonal c",
                 FadeIn(main_formula), run_time=0.8)

        # Specific substitution colour-coded
        spec = VGroup(
            MathTex(r"5^2", font_size=36, color=PINK),
            MathTex(r"=", font_size=36, color=INK),
            MathTex(r"4^2", font_size=36, color=TEAL),
            MathTex(r"+", font_size=36, color=INK),
            MathTex(r"3^2", font_size=36, color=GOLD),
        ).arrange(RIGHT, buff=0.12).move_to([3.5, 0.4, 0])

        self.cue("Here, twenty-five equals sixteen plus nine",
                 FadeIn(spec), run_time=0.8)

        self.cue("But checking one example is evidence, not a proof",
                 Indicate(main_formula, scale_factor=1.05), run_time=0.7)

        p5 = eq(r"c^2 = a^2 + b^2", 38, INK)
        p6 = txt("One example ≠ proof", 24, MUTED)
        self.cue("We still need to explain why this must work for every right triangle",
                 self.panel_to(p5, p6), run_time=0.8)

        self.finish()

        # ── BEAT 08 ──────────────────────────────────────────────────────
        self.begin("08")

        # Fade squares slightly
        self.play(
            sq_a.animate.set_opacity(0.4),
            sq_b.animate.set_opacity(0.4),
            sq_c.animate.set_opacity(0.4),
            run_time=0.6)

        # New generic triangle on left
        gA = np.array([-5.5, -1.5, 0])
        gB = np.array([-3.8, -1.5, 0])
        gC = np.array([-3.8,  0.3, 0])
        gen_tri = Polygon(gA, gB, gC,
                          stroke_color=INK, stroke_width=2,
                          fill_color=BG, fill_opacity=0)
        gen_ra  = make_right_angle_mark(gB, gA, gC, size=0.16, color=INK)
        gen_la  = txt("a", 22, TEAL).move_to((gA+gB)/2 + np.array([0,-0.2,0]))
        gen_lb  = txt("b", 22, GOLD).move_to((gB+gC)/2 + np.array([0.2,0,0]))
        gen_lc  = txt("c", 22, PINK).move_to((gA+gC)/2 + np.array([-0.2,0.1,0]))

        self.cue("A proof must work for any positive lengths a and b",
                 AnimationGroup(
                     Create(gen_tri), Create(gen_ra),
                     FadeIn(gen_la), FadeIn(gen_lb), FadeIn(gen_lc),
                     lag_ratio=0.2),
                 run_time=1.0)

        # Thought bubble text
        thought = txt("Works for ANY a and b?", 24, GOLD)
        thought.move_to([-4.65, 1.1, 0])
        thought_box = SurroundingRectangle(thought, color=GOLD,
                                           stroke_width=1.5, buff=0.15,
                                           corner_radius=0.2)

        self.cue("The claim is that the square on the hypotenuse has exactly the same area",
                 AnimationGroup(FadeIn(thought), Create(thought_box),
                                lag_ratio=0.3),
                 run_time=0.9)

        p7 = txt("Proof needed for all a, b", 26, INK)
        p8 = eq(r"c^2 = a^2 + b^2 \;?", 32, MUTED)
        self.cue("First, though, let us make sure we can identify the sides",
                 self.panel_to(p7, p8), run_time=0.8)

        self.finish()
        self.done()

        B = np.array([-5.2 + 4*SCALE, -1.8,           0.0])
        C = np.array([-5.2 + 4*SCALE, -1.8 + 3*SCALE, 0.0])

# C04 – BUILD A PROOF
# ═══════════════════════════════════════════════════════════════════════════
class C04(LessonScene):
    def construct(self):
        self.chapter(4)

        # We build the classic "four triangles in a square" proof diagram.
        # Outer square side = a+b.  We use a=4s, b=3s, s=0.38 for display.
        s = 0.38
        a = 4 * s
        b = 3 * s
        side = a + b   # outer square side

        # Outer square bottom-left corner
        OX, OY = -5.8, -2.1
        # Outer square corners (CCW from bottom-left)
        SQ = [
            np.array([OX,        OY,        0.0]),
            np.array([OX+side,   OY,        0.0]),
            np.array([OX+side,   OY+side,   0.0]),
            np.array([OX,        OY+side,   0.0]),
        ]

        # Four right triangles, each with right angle at an outer corner.
        # Triangle i has right angle at SQ[i].
        # Arrangement: legs a (teal) and b (gold) along the outer edges.
        # Bottom-left corner: leg a goes RIGHT (east), leg b goes UP (north)
        # Bottom-right corner: leg b goes UP, leg a goes LEFT
        # Top-right corner: leg a goes LEFT, leg b goes DOWN
        # Top-left corner: leg b goes DOWN, leg a goes RIGHT
        # This gives hypotenuses forming a tilted inner square.

        def tri_corners(i):
            """Return (right_angle_pt, end_of_a_leg, end_of_b_leg) for corner i."""
            P = SQ[i]
            if i == 0:  # bottom-left: a→east, b→north
                return P, P+np.array([a,0,0]), P+np.array([0,b,0])
            elif i == 1:  # bottom-right: b→north, a→west
                return P, P+np.array([0,b,0]), P+np.array([-a,0,0])
            elif i == 2:  # top-right: a→west, b→south
                return P, P+np.array([-a,0,0]), P+np.array([0,-b,0])
            else:  # top-left: b→south, a→east
                return P, P+np.array([0,-b,0]), P+np.array([a,0,0])

        def make_proof_triangle(i, fill=True):
            P, Ea, Eb = tri_corners(i)
            poly = Polygon(P, Ea, Eb,
                           stroke_color=INK, stroke_width=1.5,
                           fill_color=GRID, fill_opacity=0.5 if fill else 0)
            leg_a = Line(P, Ea, color=TEAL, stroke_width=2.5)
            leg_b = Line(P, Eb, color=GOLD, stroke_width=2.5)
            hyp   = Line(Ea, Eb, color=PINK, stroke_width=2.5)
            ra    = make_right_angle_mark(P, Ea, Eb, size=0.14, color=INK)
            return VGroup(poly, leg_a, leg_b, hyp, ra)

        # Central square corners = hypotenuse endpoints
        def central_corners():
            pts = []
            for i in range(4):
                P, Ea, Eb = tri_corners(i)
                pts.append(Ea)
            return pts

        # ── BEAT 13 ──────────────────────────────────────────────────────
        self.begin("13")

        # Draw outer square first
        outer_sq = Polygon(*SQ,
                           stroke_color=MUTED, stroke_width=2,
                           fill_color=BG, fill_opacity=0)
        self.cue("Let us build the proof",
                 Create(outer_sq), run_time=0.7)

        # Place triangles one by one
        tris = [make_proof_triangle(i) for i in range(4)]

        self.cue("Take a right triangle with legs a and b and hypotenuse c",
                 Create(tris[0]), run_time=0.8)
        self.cue("Make four identical copies",
                 AnimationGroup(Create(tris[1]), Create(tris[2]), Create(tris[3]),
                                lag_ratio=0.3),
                 run_time=1.0)

        # Label outer side a+b
        ab_lbl = eq(r"a+b", 28, INK)
        ab_lbl.move_to(np.array([OX + side/2, OY - 0.28, 0]))
        self.cue("the outer side is a plus b",
                 FadeIn(ab_lbl), run_time=0.6)

        # Label a and b on bottom triangle legs
        la_lbl = txt("a", 20, TEAL).move_to(
            tri_corners(0)[0] + (tri_corners(0)[1]-tri_corners(0)[0])*0.5
            + np.array([0, -0.2, 0]))
        lb_lbl = txt("b", 20, GOLD).move_to(
            tri_corners(0)[0] + (tri_corners(0)[2]-tri_corners(0)[0])*0.5
            + np.array([-0.2, 0, 0]))
        self.play(FadeIn(la_lbl), FadeIn(lb_lbl), run_time=0.5)

        # Central region highlight
        cc = central_corners()
        central_sq = Polygon(*cc,
                             stroke_color=PINK, stroke_width=2,
                             fill_color=PINK, fill_opacity=0.2)
        self.cue("The four hypotenuses enclose a shape in the centre",
                 Create(central_sq), run_time=0.8)

        p1 = eq(r"\text{Outer side} = a+b", 30, INK)
        p2 = txt("4 congruent triangles", 24, MUTED)
        self.cue("Around each outer edge, a leg of length a meets a leg of length b",
                 self.panel_to(p1, p2), run_time=0.8)

        self.finish()

        # ── BEAT 14 ──────────────────────────────────────────────────────
        self.begin("14")

        # Highlight central square with pink fill
        self.play(central_sq.animate.set_fill(PINK, opacity=0.35), run_time=0.5)

        cc = central_corners()
        c_labels = VGroup()
        for i in range(4):
            mid = (cc[i] + cc[(i+1)%4]) / 2
            offset_dir = mid - np.array([OX + side/2, OY + side/2, 0])
            offset_dir = offset_dir / (np.linalg.norm(offset_dir) + 1e-6) * 0.22
            c_labels.add(txt("c", 20, PINK).move_to(mid + offset_dir))

        self.cue("Is that central shape really a square",
                 FadeIn(c_labels), run_time=0.7)

        # Zoom into one corner of central square
        # Corner cc[0] is the hypotenuse endpoint of triangle 0 and triangle 3
        # The two acute angles meeting there are from tri 0 and tri 3
        corner_pt = cc[0]  # = tri_corners(0)[1]

        # Angle arcs at corner_pt
        # From tri 0: hyp goes to cc[1], leg_a goes to SQ[0]
        # From tri 3: hyp goes to cc[3], leg_b goes to SQ[0]+...
        # Actually the two triangle angles at corner_pt:
        # tri 0: angle at Ea = angle between (P0→Ea) and (Ea→Eb0)
        # tri 3: angle at Ea3 = ...
        # For display, just show alpha and beta arcs

        alpha_arc = Arc(radius=0.22, start_angle=PI,

        self.cue("Two acute triangle angles sit beside it along a straight outer edge",

        # ── BEAT 15 ──────────────────────────────────────────────────────
        self.begin("15")

        # Show alpha+beta=90 in a triangle
        angle_sum = eq(r"\alpha + \beta = 90^\circ", 34, INK).move_to([3.5, 1.6, 0])
        self.cue("The angles inside a triangle total one hundred and eighty degrees",
                 FadeIn(angle_sum), run_time=0.7)

        theta_arc = Arc(radius=0.18, start_angle=math.radians(37),
                        angle=math.radians(90),

        theta_lbl = MathTex(r"\theta", font_size=22, color=INK)
        theta_lbl.move_to(corner_pt + np.array([0.1, 0.32, 0]))

        self.cue("Our triangle already has a ninety-degree angle, so alpha and beta together total ninety degrees",
                 AnimationGroup(Create(theta_arc), FadeIn(theta_lbl),

        self.play(central_sq.animate.set_stroke(GOLD, width=3), run_time=0.5)
