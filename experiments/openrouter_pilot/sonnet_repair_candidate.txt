```python
from manim import *
import numpy as np
import math


class SlopeInterceptDemo(Scene):
    def construct(self):
        # ── Layout constants ──────────────────────────────────────────────
        # Equal unit scale: x-range=6 units, y-range=8 units
        # Choose length so units match: 4.8 units length for x (6 units → 0.8/unit)
        # y: 6.4 units length for 8 units → 0.8/unit  ✓
        X_LEN = 4.8   # 6 units wide  → 0.8 scene-units per data-unit
        Y_LEN = 6.4   # 8 units tall  → 0.8 scene-units per data-unit

        # Left panel: axes centred so top of axes ≤ top of frame and
        # bottom of axes stays above y = -2.6 (caption region).
        # Axes centre in data coords: x_mid=0, y_mid=1  (range -3..5)
        # Scene position: keep axes left of centre.
        AX_ORIGIN = LEFT * 3.0 + UP * 0.5   # scene coords of data origin (0,0)

        # Right panel: fixed x = 3.5, rows from top downward
        RP_X = 3.5
        RP_TOP = 3.2          # top row y in scene coords
        RP_ROW = 0.58         # row pitch (scene units)
        MIN_FS = 22           # minimum font size

        # ── Axes ──────────────────────────────────────────────────────────
        axes = Axes(
            x_range=[-3, 3, 1],
            y_range=[-3, 5, 1],
            x_length=X_LEN,
            y_length=Y_LEN,
            axis_config={
                "include_numbers": True,
                "numbers_to_include": [-3, -2, -1, 0, 1, 2, 3],
                "font_size": MIN_FS,
                "tip_length": 0.18,
            },
            y_axis_config={
                "numbers_to_include": [-3, -2, -1, 0, 1, 2, 3, 4, 5],
            },
        ).move_to(AX_ORIGIN)

        # Verify bottom of axes is above caption boundary (-2.6 scene y).
        # axes centre is at AX_ORIGIN.  y data -3 maps to scene y:
        #   AX_ORIGIN.y + (-3 - 1) * (Y_LEN/8)  [y_mid=1 for range -3..5]
        # = 0.5 + (-4)*0.8 = 0.5 - 3.2 = -2.7  → shift up slightly
        # Recalculate: data centre = (-3+5)/2 = 1, so axes.move_to places
        # data point (0,1) at AX_ORIGIN.  Bottom (y=-3) is at
        # AX_ORIGIN.y - (1-(-3))*0.8 = 0.5 - 3.2 = -2.7.
        # Shift AX_ORIGIN up by 0.15 to clear -2.6.
        AX_ORIGIN = LEFT * 3.0 + UP * 0.65
        axes.move_to(AX_ORIGIN)

        x_label = axes.get_x_axis_label(
            MathTex("x", font_size=26), direction=RIGHT)
        y_label = axes.get_y_axis_label(
            MathTex("y", font_size=26), direction=UP)

        # ── Right-panel row helper ────────────────────────────────────────
        def rp_pos(row):
            """Scene position for right-panel row (row 0 = topmost)."""
            return np.array([RP_X, RP_TOP - row * RP_ROW, 0])

        # ── Equation builder ─────────────────────────────────────────────
        def make_eq(m_val, b_val):
            if m_val == 1:
                m_part = ""
            elif m_val == -1:
                m_part = "-"
            elif m_val == int(m_val):
                m_part = f"{int(m_val)}"
            else:
                m_part = f"{m_val:.1f}"

            b_sign = "+" if b_val >= 0 else "-"
            b_abs = abs(b_val)
            if b_abs == int(b_abs):
                b_str = f"{int(b_abs)}"
            else:
                b_str = f"{b_abs:.1f}"

            if m_val == 1:
                eq_str = rf"y = x {b_sign} {b_str}"
            elif m_val == -1:
                eq_str = rf"y = -x {b_sign} {b_str}"
            else:
                eq_str = rf"y = {m_part}x {b_sign} {b_str}"

            return MathTex(eq_str, font_size=34).move_to(rp_pos(0))

        # ── Right-panel text label ────────────────────────────────────────
        def make_label(text_str, row, fs=24):
            fs = max(fs, MIN_FS)
            t = Text(text_str, font_size=fs)
            t.move_to(rp_pos(row))
            return t

        def make_math_label(tex_str, row, fs=28):
            fs = max(fs, MIN_FS)
            t = MathTex(tex_str, font_size=fs)
            t.move_to(rp_pos(row))
            return t

        # ── Clip line to axis bounds ──────────────────────────────────────
        def clipped_line(m, b, color=YELLOW):
            x0, x1 = -3.0, 3.0
            y0, y1 = -3.0, 5.0
            pts = []
            # left edge
            y = m * x0 + b
            if y0 - 1e-9 <= y <= y1 + 1e-9:
                pts.append((x0, float(np.clip(y, y0, y1))))
            # right edge
            y = m * x1 + b
            if y0 - 1e-9 <= y <= y1 + 1e-9:
                pts.append((x1, float(np.clip(y, y0, y1))))
            # bottom edge
            if abs(m) > 1e-9:
                x = (y0 - b) / m
                if x0 + 1e-9 < x < x1 - 1e-9:
                    pts.append((x, y0))
            # top edge
            if abs(m) > 1e-9:
                x = (y1 - b) / m
                if x0 + 1e-9 < x < x1 - 1e-9:
                    pts.append((x, y1))
            # horizontal special cases
            if abs(m) < 1e-9:
                if y0 <= b <= y1:
                    pts = [(x0, b), (x1, b)]
            # deduplicate
            unique = []
            for p in pts:
                if not any(
                    abs(p[0] - q[0]) < 1e-9 and abs(p[1] - q[1]) < 1e-9
                    for q in unique
                ):
                    unique.append(p)
            if len(unique) < 2:
                # fallback: just draw between x endpoints
                return Line(
                    axes.c2p(x0, m * x0 + b),
                    axes.c2p(x1, m * x1 + b),
                    color=color, stroke_width=3,
                )
            unique.sort(key=lambda p: p[0])
            return Line(
                axes.c2p(unique[0][0], unique[0][1]),
                axes.c2p(unique[-1][0], unique[-1][1]),
                color=color, stroke_width=3,
            )

        # ── Slope triangle builder ────────────────────────────────────────
        def make_slope_triangle(m, b, x_base=0):
            """
            Build rise/run triangle anchored at (x_base, m*x_base+b).
            run = 1 step right, rise = m (signed).
            Returns (tri_run, tri_rise, lbl_run, lbl_rise).
            """
            y_base = m * x_base + b
            x_tip = x_base + 1
            y_tip = y_base + m   # = m*(x_base+1)+b

            p_start = axes.c2p(x_base, y_base)
            p_run   = axes.c2p(x_tip,  y_base)
            p_end   = axes.c2p(x_tip,  y_tip)

            tr = Line(p_start, p_run,  color=GREEN,  stroke_width=3)
            tr2 = Line(p_run,  p_end,  color=ORANGE, stroke_width=3)

            # run label: below the horizontal leg
            run_mid = axes.c2p(x_base + 0.5, y_base)
            lbl_r = MathTex(r"\Delta x=1", font_size=24, color=GREEN).next_to(
                run_mid, DOWN, buff=0.14)

            # rise label: right of the vertical leg
            rise_mid = axes.c2p(x_tip, y_base + m / 2)
            rise_val = int(m) if m == int(m) else f"{m:.1f}"
            sign = "+" if m >= 0 else ""
            lbl_rise_str = rf"\Delta y={sign}{rise_val}"
            lbl_ri = MathTex(lbl_rise_str, font_size=24, color=ORANGE).next_to(
                rise_mid, RIGHT, buff=0.14)

            return tr, tr2, lbl_r, lbl_ri

        # ═══════════════════════════════════════════════════════════════════
        # STAGE A  (0 – 32 s)
        # ═══════════════════════════════════════════════════════════════════

        # 0-3 s  draw axes
        self.play(Create(axes), Write(x_label), Write(y_label), run_time=3)

        # 3-5 s  show initial equation y = x + 1
        eq_A = make_eq(1, 1)
        self.play(Write(eq_A), run_time=2)

        # 5-8 s  draw line y=x+1
        line_A = clipped_line(1, 1)
        self.play(Create(line_A), run_time=3)

        # 8-11 s  mark y-intercept (0,1)
        dot_b = Dot(axes.c2p(0, 1), color=RED, radius=0.09)
        lbl_b = MathTex("(0,1)", font_size=26, color=RED).next_to(
            dot_b, RIGHT, buff=0.15)
        # row 1: b info (row 0 is equation)
        info1 = make_label("y-intercept  b = 1", row=1)
        self.play(FadeIn(dot_b), Write(lbl_b), Write(info1), run_time=3)

        # 11-13 s  pause
        self.wait(2)

        # 13-15 s  show slope=1 label  (row 2)
        info2 = make_label("slope  m = 1", row=2)
        self.play(Write(info2), run_time=2)

        # 15-18 s  draw rise/run triangle for m=1 on y=x+1
        tri_run, tri_rise, lbl_run, lbl_rise = make_slope_triangle(1, 1, x_base=0)
        self.play(
            Create(tri_run), Create(tri_rise),
            Write(lbl_run), Write(lbl_rise),
            run_time=3,
        )

        # 18-20 s  pause
        self.wait(2)

        # 20-22 s  announce b change  (row 3)
        info3 = make_label("increase b: line shifts up", row=3)
        self.play(Write(info3), run_time=1)

        # 22-25 s  animate b: 1 → 2
        # New targets: line, equation, dot, dot-label, info1, triangle
        line_A2 = clipped_line(1, 2)
        eq_A2   = make_eq(1, 2)
        dot_b2  = Dot(axes.c2p(0, 2), color=RED, radius=0.09)
        lbl_b2  = MathTex("(0,2)", font_size=26, color=RED).next_to(
            dot_b2, RIGHT, buff=0.15)
        info1_b2 = make_label("y-intercept  b = 2", row=1)

        # New triangle anchored to new line y=x+2 at x=0
        tri_run2, tri_rise2, lbl_run2, lbl_rise2 = make_slope_triangle(1, 2, x_base=0)

        self.play(
            Transform(line_A,  line_A2),
            Transform(eq_A,    eq_A2),
            Transform(dot_b,   dot_b2),
            Transform(lbl_b,   lbl_b2),
            Transform(info1,   info1_b2),   # FIX #1: b-state label updates with line
            Transform(tri_run,  tri_run2),  # FIX #1: triangle moves with line
            Transform(tri_rise, tri_rise2),
            Transform(lbl_run,  lbl_run2),
            Transform(lbl_rise, lbl_rise2),
            run_time=3,
        )

        # 25-27 s  slope-still label  (row 4 — below info3 at row 3)
        info4 = make_label("slope still  m = 1", row=4)
        self.play(Write(info4), run_time=2)

        # 27-30 s  pause
        self.wait(3)

        # ═══════════════════════════════════════════════════════════════════
        # TRANSITION  (30 – 33 s)
        # ═══════════════════════════════════════════════════════════════════

        self.play(
            FadeOut(info1), FadeOut(info2), FadeOut(info3), FadeOut(info4),
            FadeOut(tri_run), FadeOut(tri_rise),
            FadeOut(lbl_run), FadeOut(lbl_rise),
            FadeOut(lbl_b), FadeOut(dot_b),
            run_time=2,
        )

        # Reset line to b=1, m=1
        line_B_init = clipped_line(1, 1)
        eq_B        = make_eq(1, 1)
        dot_pivot   = Dot(axes.c2p(0, 1), color=RED, radius=0.09)
        lbl_pivot   = MathTex("(0,1)", font_size=26, color=RED).next_to(
            dot_pivot, RIGHT, buff=0.15)

        self.play(
            Transform(line_A, line_B_init),
            Transform(eq_A,   eq_B),
            FadeIn(dot_pivot), Write(lbl_pivot),
            run_time=1,
        )

        # ═══════════════════════════════════════════════════════════════════
        # STAGE B  (33 – 60 s)
        # ═══════════════════════════════════════════════════════════════════

        # Row layout for Stage B (row 0 = equation, always):
        #   row 0 : equation  (make_eq places itself at rp_pos(0))
        #   row 1 : title_B
        #   row 2 : info_B1  (b fixed)
        #   row 3 : current-m label
        #   row 4 : slope_frac  (fraction display)
        #   row 5 : summary1
        #   row 6 : summary2

        # 33-35 s  Stage B title
        title_B = make_label("Stage B: vary slope m", row=1)
        self.play(Write(title_B), run_time=2)

        # 35-37 s  b=1 fixed note
        info_B1 = make_label("b = 1 fixed, pivot (0,1)", row=2)
        self.play(Write(info_B1), run_time=2)

        # 37-40 s  m=1 → m=0
        line_m0 = clipped_line(0, 1)
        eq_m0   = make_eq(0, 1)
        info_m0 = make_label("m = 0  (horizontal)", row=3)
        self.play(
            Transform(line_A, line_m0),
            Transform(eq_A,   eq_m0),
            Write(info_m0),
            run_time=3,
        )
        self.wait(1)

        # 41-44 s  m=0 → m=-1
        line_m1n = clipped_line(-1, 1)
        eq_m1n   = make_eq(-1, 1)
        info_m1n = make_label("m = -1  (negative slope)", row=3)
        self.play(
            Transform(line_A, line_m1n),
            Transform(eq_A,   eq_m1n),
            Transform(info_m0, info_m1n),   # replace in same row
            run_time=3,
        )

        # 44-46 s  pause
        self.wait(2)

        # 46-48 s  draw signed rise/run triangle for m=-1
        # Anchor at x=0, y=1 on line y=-x+1
        p0 = axes.c2p(0, 1)
        p1 = axes.c2p(1, 1)
        p2 = axes.c2p(1, 0)
        tri2_run  = Line(p0, p1, color=GREEN,  stroke_width=3)
        tri2_rise = Line(p1, p2, color=ORANGE, stroke_width=3)

        # run label: above the horizontal leg (avoids line overlap)
        lbl2_run = MathTex(r"\Delta x=1", font_size=24, color=GREEN).next_to(
            axes.c2p(0.5, 1), UP, buff=0.14)
        # rise label: right of the vertical leg
        lbl2_rise = MathTex(r"\Delta y=-1", font_size=24, color=ORANGE).next_to(
            axes.c2p(1, 0.5), RIGHT, buff=0.14)

        self.play(
            Create(tri2_run), Create(tri2_rise),
            Write(lbl2_run), Write(lbl2_rise),
            run_time=2,
        )

        # 48-50 s  slope fraction — row 4, no overlap with info_m1n (row 3)
        # FIX #2: place slope_frac at row 4, well below info_m1n at row 3
        slope_frac = make_math_label(
            r"m = \frac{\Delta y}{\Delta x} = \frac{-1}{1} = -1",
            row=4, fs=26,
        )
        self.play(Write(slope_frac), run_time=2)

        # 50-53 s  pause
        self.wait(3)

        # 53-55 s  summary1 at row 5
        summary1 = make_label("b shifts line vertically", row=5)
        self.play(Write(summary1), run_time=1.5)

        # 55-57 s  summary2 at row 6  (FIX #2: explicit row, no minus-DOWN trick)
        summary2 = Text(
            "m controls steepness & direction",
            font_size=24,
        ).move_to(rp_pos(6))
        self.play(Write(summary2), run_time=1.5)

        # 57-60 s  final pause — preserve graph and all labels  (FIX #4: 60 s total)
        self.wait(3)
```
